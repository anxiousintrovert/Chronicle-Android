cmake_minimum_required(VERSION 3.20)
project(chronicle_native_probe C)
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Build Chronicle plugins with Clang")
endif()
add_library(probe SHARED probe.c)
target_include_directories(probe PRIVATE ../../port/src)
set_target_properties(probe PROPERTIES C_STANDARD 99 PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/package/mod.json" CONTENT "{\n  \"schema_version\":1,\n  \"api_version\":1,\n  \"id\":\"example.native-probe\",\n  \"name\":\"Native Probe\",\n  \"version\":\"1.0.0\",\n  \"game\":\"chronicle\",\n  \"requires_capabilities\":[\"plugins.native.v1\"],\n  \"permissions\":[\"native.extension\"],\n  \"native_library\":\"$<TARGET_FILE_NAME:probe>\",\n  \"files\":{}\n}\n")
