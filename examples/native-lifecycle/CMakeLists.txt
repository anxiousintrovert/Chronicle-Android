cmake_minimum_required(VERSION 3.20)
project(chronicle_native_lifecycle C)
if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "Build Chronicle plugins with Clang")
endif()
add_library(lifecycle SHARED lifecycle.c)
target_include_directories(lifecycle PRIVATE ../../port/src)
set_target_properties(lifecycle PROPERTIES C_STANDARD 99 PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package")
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/package/mod.json" CONTENT "{\n  \"schema_version\":1,\n  \"api_version\":1,\n  \"id\":\"example.native-lifecycle\",\n  \"name\":\"Native Lifecycle\",\n  \"version\":\"1.0.0\",\n  \"game\":\"chronicle\",\n  \"requires_capabilities\":[\"plugins.native.v2\"],\n  \"permissions\":[\"native.extension\"],\n  \"native_library\":\"$<TARGET_FILE_NAME:lifecycle>\",\n  \"files\":{}\n}\n")
